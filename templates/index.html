Grammar Reference

            
program     = fn_decl*
fn_decl     = "fn" ID "(" params ")"
              ("->" type)? block
params      = param ("," param)* | ε
param       = "mut"? ID ":" type
block       = "{" stmt* "}"
stmt        = let | return | if
            | while | for | assign
let_stmt    = "let" "mut"? ID
              (":" type)? ("=" expr)? ";"
if_stmt     = "if" expr block
              ("else" (block|if))?
while_stmt  = "while" expr block
for_stmt    = "for" ID "in" expr block
expr        = additive (comp additive)*
type        = "i32" | "[" type ";" NUM "]"
